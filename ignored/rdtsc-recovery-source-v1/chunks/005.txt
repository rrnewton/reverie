+/// CR4 is updated in place; unrelated execution and paging bits are retained.
+pub(crate) fn set_userspace_rdtsc_interception(vcpu: &VcpuFd, enabled: bool) -> Result<()> {
+    let mut sregs = vcpu.get_sregs()?;
