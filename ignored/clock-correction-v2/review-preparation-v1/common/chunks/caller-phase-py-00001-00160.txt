#!/usr/bin/python3
"""Bounded Reverie component phases using the unchanged reviewed observer."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import stat
import subprocess
import sys
import time

import cache_lease
from common import bounded_process, check_file, digest, file_record, json_read, read, require, terminal, write_new

HERE = Path(__file__).resolve().parent
OWNER = HERE.parents[2]
REPO = Path(json_read(HERE/'SETUP.json')['source_root'])
require(REPO == OWNER or REPO.is_relative_to(OWNER/'ignored'), 'source root outside owned slot')


def check(plan):
    require(len(plan['sources']) == 1, 'exactly one selected Reverie snapshot must be bound')
    require({entry['root'] for entry in plan['sources']} == set(json_read(HERE/'SETUP.json')['composed_sources']), 'composed source roots differ')
    for entry in plan['sources']:
        root = Path(entry['root'])
        require(root.is_relative_to(OWNER/'ignored'), 'composed source escaped owned tree')
        check_file(entry['manifest'])
        for row in json_read(entry['manifest']['path']):
            p = root / row['path']
            require(not Path(row['path']).is_absolute() and '..' not in Path(row['path']).parts, 'invalid source path')
            if row['mode'] == '120000':
                require(p.is_symlink() and hashlib.sha256(os.fsencode(os.readlink(p))).hexdigest() == row['sha256'], 'source symlink changed')
            else:
                require(row['mode'] in ('100644', '100755') and p.is_file() and not p.is_symlink(), 'source type changed')
                require(bool(p.stat().st_mode & 0o111) == (row['mode'] == '100755'), 'source mode changed')
                require(digest(p) == row['sha256'], 'source changed: ' + str(p))
    for entry in plan.get('empty_directories', []):
        p = Path(entry['path']); info = p.lstat()
        require(stat.S_ISDIR(info.st_mode) and stat.S_IMODE(info.st_mode) == 0o700 and [info.st_dev, info.st_ino, info.st_uid] == entry['identity'], 'guest directory identity changed')
        require(not list(p.iterdir()), 'guest HOME/config contents changed')
    for row in plan['inputs']:
        check_file(row)
    for link in plan.get('input_symlinks', []):
        p = Path(link['path'])
        require(p.is_symlink() and os.readlink(p) == link['target'], 'loader alias changed: ' + str(p))
    for name in plan['absent_inputs']:
        require(not Path(name).exists() and not Path(name).is_symlink(), 'new Cargo configuration: ' + name)
    p = Path(plan['target']['path'])
    s = p.stat()
    require([s.st_dev,s.st_ino,s.st_uid] == plan['target']['identity'] and not p.is_symlink(), 'owned target changed')
    if plan.get('dependencies'):
        check_file(plan['dependencies'])
        for item in json_read(plan['dependencies']['path']):
            check_file(item)


def scm(plan, prefix):
    state = {}
    for name, args in [('head',['rev-parse','HEAD']),('branch',['branch','--show-current']),('index',['ls-files','--stage'])]:
        p = subprocess.run(['/usr/bin/git','-C',str(OWNER),*args],env=plan['environment'],capture_output=True,timeout=30,check=True)
        state[name] = hashlib.sha256(p.stdout).hexdigest() if name == 'index' else p.stdout.decode().strip()
    write_new(prefix,state)
    require(state == plan['scm'],'SCM state changed')


def payload(plan, sha):
    fd = cache_lease.join(plan['lease'], sha)
    try:
        check(plan)
        for k,v in plan['environment'].items():
            require(os.environ.get(k) == v, 'payload environment changed: '+k)
        extras = [k for k in os.environ if k not in plan['environment'] and (k.startswith(('CARGO_','RUST','LD_','NEXTEST_','REVERIE_','THIRD_PARTY_','HERMIT_','PKG_CONFIG_')) or k in ('CC','CXX','CFLAGS','CPPFLAGS','CXXFLAGS','LDFLAGS'))]
        require(not extras, 'undeclared overrides: '+str(extras))
        group = read('/proc/self/cgroup',4096).decode()
        require('/safehermit-' in group and group.strip().endswith('.service'),'payload is not in observed service')
        out = Path(plan['output'])
        write_new(out/'payload.json',dict(argv=plan['argv'],cwd=str(REPO),environment=dict(os.environ),cgroup=group,pid=os.getpid(),plan_sha256=sha))
        for relative in plan.get('owned_output_directories', []):
            rel = Path(relative)
            require(not rel.is_absolute() and '..' not in rel.parts and len(rel.parts) == 1, 'invalid output directory')
            path = out / rel
            require(not path.exists() and not path.is_symlink(), 'retain previous output directory')
            path.mkdir(mode=0o700)
        started = time.monotonic()
        child = subprocess.Popen(plan['argv'],cwd=REPO,stdin=subprocess.DEVNULL)
        code = child.wait()  # Actual observer owns complete aggregate lifetime.
        write_new(out/'payload-exit.json',dict(returncode=code,reaped=True,elapsed_seconds=time.monotonic()-started))
        check(plan)
        cache_lease.check_held(fd,plan['lease'],sha)
        return code if code >= 0 else 128-code
    finally:
        os.close(fd)


def inspect(plan, raw):
    kind = plan['kind']
    if kind == 'format':
        require(not raw.strip(),'rustfmt produced differences')
        return dict(format_check=True,changed_rust_paths=len(plan['argv'])-6)
    if kind == 'metadata':
        value = json.loads(raw)
        require(value['workspace_root'] == str(REPO),'metadata workspace changed')
        setup = json_read(HERE/'SETUP.json')
        for package in value['packages']:
            if package['name'] in setup['reverie_packages']:
                require(package['source'] is None and Path(package['manifest_path']).is_relative_to(Path(setup['reverie_root'])), 'resolved Reverie escaped composed source')
        return dict(package_count=len(value['packages']),workspace_root=value['workspace_root'])
    if kind in ('compile','check'):
        rows = [json.loads(x) for x in raw.splitlines()]
        require([r['success'] for r in rows if r.get('reason')=='build-finished'] == [True], 'missing successful Cargo completion')
        diagnostics = [r for r in rows if r.get('reason') == 'compiler-message']
        write_new(Path(plan['control'])/'compiler-diagnostics.json',diagnostics)
        require(not diagnostics,'structured compiler diagnostics require disposition')
        if kind == 'check':
            return dict(cargo_complete=True,compiler_diagnostics=0)
        artifacts = {}
        for selector in plan['artifact_selectors']:
            require(type(selector['test_profile']) is bool, 'artifact test profile must be exact bool')
            require(Path(selector['manifest_path']) == REPO / selector['manifest_relative'], 'artifact manifest outside selected source')
            require(not Path(selector['manifest_relative']).is_absolute() and '..' not in Path(selector['manifest_relative']).parts, 'invalid artifact manifest path')
            matches = [r for r in rows if r.get('reason')=='compiler-artifact'
                       and r.get('manifest_path')==selector['manifest_path']
                       and r.get('target',{}).get('name')==selector['target']
                       and r.get('target',{}).get('kind')==selector['kind']
                       and r.get('profile',{}).get('test') is selector['test_profile'] and r.get('executable')]
            require(len(matches)==1,'Cargo must emit exactly one selected artifact: '+selector['id'])
            path=Path(matches[0]['executable'])
            require(path.resolve()==path and path.is_relative_to(Path(plan['target']['path']))
                    and not path.is_symlink() and os.access(path,os.X_OK),'test artifact escaped owned target')
            with path.open('rb') as elf: require(elf.read(4)==b'\x7fELF','artifact is not ELF')
            artifacts[selector['id']]=dict(cargo_artifact=matches[0],file=file_record(path))
        write_new(Path(plan['control'])/'artifacts.json',artifacts)
        return dict(artifacts=artifacts,compiler_diagnostics=0)
    if kind == 'guest':
        from guest_readback import inspect_guest
        return inspect_guest(plan, raw)
    if kind == 'comparison':
        from guest_readback import inspect_comparison
        return inspect_comparison(plan)
    rows = [json.loads(x) for x in raw.splitlines()]
    if kind == 'list':
        names = [r['name'] for r in rows if r.get('type')=='test']
        require(len(names)==len(set(names)),'duplicate test names')
        require(all(names.count(x)==1 for x in plan['selected']),'missing selected test')
        return dict(names=names,count=len(names),selected=plan['selected'])
    require(kind=='test','unknown result type')
    started = [r['name'] for r in rows if r.get('type')=='test' and r.get('event')=='started']
    ended = [r for r in rows if r.get('type')=='test' and r.get('event') in ('ok','failed','ignored')]
    suites = [r for r in rows if r.get('type')=='suite' and r.get('event') in ('ok','failed')]
    write_new(Path(plan['control'])/'outcomes.json',dict(started=started,ended=ended,suites=suites))
    require(started == plan['selected'] and len(ended)==len(plan['selected']) and [r['name'] for r in ended]==plan['selected'] and all(r['event']=='ok' for r in ended),'actual selected outcomes differ')
    require(len(suites)==1 and suites[0]['event']=='ok' and suites[0]['passed']==len(plan['selected']) and suites[0]['failed']==suites[0]['ignored']==suites[0]['measured']==0,'test count/failure/skip mismatch')
    return dict(executed=len(ended),summary=suites[0],names=started)


def launch(plan, sha):
    fd = cache_lease.claim(plan['lease'],sha)
    control = Path(plan['control']);control.mkdir(mode=0o700)
