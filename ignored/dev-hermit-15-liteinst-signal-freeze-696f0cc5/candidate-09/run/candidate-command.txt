verify_file() {
    local path="$1"
    local expected_hash="$2"
    local expected_mode="$3"
    local canonical
    local digest_line
    local digest
    local mode

    [ -f "$path" ] && [ ! -L "$path" ] || return 1
    canonical=$(realpath -e -- "$path") || return 1
    [ "$canonical" = "$path" ] || return 1
    digest_line=$(sha256sum -- "$path") || return 1
    digest=${digest_line%% *}
    [ "$digest" = "$expected_hash" ] || return 1
    mode=$(stat -c '%a' -- "$path") || return 1
    [ "$mode" = "$expected_mode" ] || return 1
}
