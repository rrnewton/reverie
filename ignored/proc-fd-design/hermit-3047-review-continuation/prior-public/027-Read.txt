12	/// The namespace used by the backend's ordinary host filesystem operations.
13	/// Namespace-changing guest syscalls are unsupported, so fork and exec retain
14	/// this snapshot. The Tool remains responsible for deterministic normalization.
15	#[derive(Debug)]
16	pub(crate) struct ProcMountSnapshot {
17	    pub(crate) mountinfo: Vec<u8>,
18	    pub(crate) mounts: Vec<u8>,
19	}

23	        Ok(Self {
24	            mountinfo: read_snapshot("/proc/self/mountinfo")?,
25	            mounts: read_snapshot("/proc/self/mounts")?,
26	        })
27	    }
28	}

30	fn read_snapshot(path: &str) -> io::Result<Vec<u8>> {
31	    // A backend capacity refusal must not become a truncated mount table.
32	    const MAX_SNAPSHOT_BYTES: u64 = 16 * 1024 * 1024;
33	    read_snapshot_bytes(std::fs::File::open(path)?, MAX_SNAPSHOT_BYTES, path)
34	}
35	
36	fn read_snapshot_bytes(source: impl Read, limit: u64, path: &str) -> io::Result<Vec<u8>> {
37	    let mut bytes = Vec::new();
38	    source.take(limit + 1).read_to_end(&mut bytes)?;
39	    if bytes.is_empty() || bytes.len() as u64 > limit {
40	        return Err(io::Error::new(
41	            io::ErrorKind::InvalidData,
42	            format!("{path} is empty or exceeds the backend mount snapshot capacity"),
43	        ));
44	    }
45	    Ok(bytes)
46	}
