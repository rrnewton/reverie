688	    fn discover_fd_from_current_process(&mut self, owner: DetTid, fd: RawFd) -> Result<(), Errno> {
689	        if self.file_handles.contains_key(&fd) {
690	            return Ok(());
691	        }
692	
693	        let fd_flags = unsafe { libc::fcntl(fd, libc::F_GETFD) };
694	        let status_flags = unsafe { libc::fcntl(fd, libc::F_GETFL) };
695	        if fd_flags == -1 || status_flags == -1 {
696	            return Err(Errno::last());
697	        }
698	        let raw_stat =
699	            stat::fstat(unsafe { BorrowedFd::borrow_raw(fd) }).map_err(|_| Errno::last())?;
700	        let file_type = stat::SFlag::from_bits_truncate(raw_stat.st_mode);
701	        let ty = if file_type.contains(stat::SFlag::S_IFIFO) {
702	            FdType::Pipe
703	        } else if file_type.contains(stat::SFlag::S_IFSOCK) {
704	            FdType::Socket
705	        } else {
706	            FdType::Regular
707	        };
708	        let mut flags = OFlag::from_bits_truncate(status_flags);
709	        let physically_nonblocking = flags.contains(OFlag::O_NONBLOCK);
710	        // Discovered descriptors have unknown provenance, so an observed
711	        // O_NONBLOCK bit must remain guest-visible. Detcore-created scheduler
712	        // pipes are registered when created and do not reach this fallback.
713	        if fd_flags & libc::FD_CLOEXEC != 0 {
714	            flags.insert(OFlag::O_CLOEXEC);
715	        }
716	        self.add_fd(owner, fd, flags, ty, Some(raw_stat.into()))?;
717	        self.with_detfd(fd, |detfd| detfd.forget_flock_mode())?;
718	        if let Some(resource) = stdio_resource(fd) {
719	            self.with_detfd(fd, |detfd| detfd.set_resource(resource.clone()))?;
720	        }
721	        if ty == FdType::Pipe && physically_nonblocking {
722	            self.with_detfd(fd, |detfd| detfd.set_physically_nonblocking())?;
723	        }
