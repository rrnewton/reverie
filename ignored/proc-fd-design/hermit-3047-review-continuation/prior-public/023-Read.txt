322	fn canonical_mount_target(path: &Path) -> PathBuf {
323	    for ancestor in path.ancestors() {
324	        if let Ok(mut canonical) = fs::canonicalize(ancestor)
325	            && let Ok(suffix) = path.strip_prefix(ancestor)
326	        {
327	            canonical.push(suffix);
328	            return canonical;
329	        }
330	    }
331	    path.to_path_buf()
332	}
